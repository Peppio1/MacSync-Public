import Foundation
import SQLite3
import XCTest
@testable import MacSyncCore

final class RetentionWriteBenchmarkTests:XCTestCase {
    func testIdenticalQueueWriteWorkload()throws {
        let fm=FileManager.default
        let root=fm.temporaryDirectory.appendingPathComponent(UUID().uuidString).resolvingSymlinksInPath()
        try fm.createDirectory(at:root,withIntermediateDirectories:true)
        defer {try? fm.removeItem(at:root)}
        let path=root.appendingPathComponent("queue.sqlite").path
        let journal=try ChangeJournal(path:path)
        var raw:OpaquePointer?
        XCTAssertEqual(sqlite3_open(path,&raw),SQLITE_OK)
        let db=try XCTUnwrap(raw)
        defer {sqlite3_close(db)}
        sqlite3_busy_timeout(db,1000)
        func exec(_ sql:String)throws {
            guard sqlite3_exec(db,sql,nil,nil,nil)==SQLITE_OK else {throw JournalError.database(String(cString:sqlite3_errmsg(db)))}
        }
        let n=ProcessInfo.processInfo.environment["MACSYNC_RETENTION_ROWS"]=="200000" ? 200000 : 10000
        let start=ProcessInfo.processInfo.systemUptime
        try exec("""
            BEGIN IMMEDIATE;
            WITH RECURSIVE seq(n) AS (SELECT 1 UNION ALL SELECT n+1 FROM seq WHERE n<\(n))
            INSERT INTO change_queue(root_id,relative_path,kind,recorded_at,completed,origin)
            SELECT 'demo','photo-'||((n-1)%\(n/2))||'.jpg','upsert',1,0,'source_monitor' FROM seq;
            COMMIT;
            """)
        let insert=ProcessInfo.processInfo.systemUptime-start
        let ackStart=ProcessInfo.processInfo.systemUptime
        try exec("BEGIN IMMEDIATE; UPDATE change_queue SET completed=1; COMMIT;")
        let ack=ProcessInfo.processInfo.systemUptime-ackStart
        XCTAssertEqual(try journal.queueSequence(),Int64(n))
        XCTAssertTrue(try journal.pending().isEmpty)
        let bytes=(try fm.attributesOfItem(atPath:path)[.size] as? NSNumber)?.int64Value ?? -1
        let wal=(try? fm.attributesOfItem(atPath:path+"-wal")[.size] as? NSNumber)?.int64Value ?? -1
        try ArchiveDatabaseSafety.verify(URL(fileURLWithPath:path),readOnly:true)
        print("RETENTION_WRITE_BENCH rows=\(n) insertSeconds=\(insert) completionSeconds=\(ack) dbBytes=\(bytes) walBytes=\(wal)")
    }
}
