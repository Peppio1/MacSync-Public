import Foundation
import XCTest
@testable import MacSyncCore

final class RetentionPredecessorReadTests:XCTestCase {
    func testReadAndContinueFromMaintainedDatabase()throws {
        let path=try XCTUnwrap(ProcessInfo.processInfo.environment["MACSYNC_RETENTION_COMPAT_DIR"])+"/retention-backup.sqlite"
        let url=URL(fileURLWithPath:path)
        try SignedRollbackTransaction.validateCompatibility(database:url)
        let journal=try ChangeJournal(path:path)
        XCTAssertEqual(try journal.queueSequence(),200000)
        XCTAssertTrue(try journal.pending().isEmpty)
        let id=try journal.enqueue(rootID:"demo",relativePath:"after-rollback.jpg",kind:.upsert)
        XCTAssertEqual(id,200001)
        XCTAssertEqual(try journal.pending().map(\.id),[id])
        try journal.markCompleted(id:id)
        XCTAssertTrue(try journal.pending().isEmpty)
        try ArchiveDatabaseSafety.verify(url,readOnly:true)
        print("RETENTION_PREDECESSOR_READ_WRITE_OK")
    }
}
